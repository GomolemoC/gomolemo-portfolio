University Academic Management Database

MySQL Database Design SQL

Project Overview

A comprehensive MySQL database system designed to manage university academic operations, including student records, course management, tutor allocations, and academic performance tracking.

Key Features

  • Student enrollment and registration management
  • Course and curriculum management with prerequisites
  • Tutor allocation and workload tracking
  • Assessment and grading system with CGPA calculation
  • Attendance tracking with automated alerts
  • Department-wise analytics and reporting

Database Schema Sample

-- Core Tables CREATE TABLE departments ( department_id INT PRIMARY KEY AUTO_INCREMENT, department_code VARCHAR(10) UNIQUE NOT NULL, department_name VARCHAR(100) NOT NULL, head_of_department INT, established_year YEAR ); CREATE TABLE students ( student_id INT PRIMARY KEY AUTO_INCREMENT, registration_number VARCHAR(20) UNIQUE NOT NULL, first_name VARCHAR(50) NOT NULL, last_name VARCHAR(50) NOT NULL, email VARCHAR(100) UNIQUE NOT NULL, enrollment_date DATE NOT NULL, current_semester INT DEFAULT 1, department_id INT, cgpa DECIMAL(4,2) DEFAULT 0.00, FOREIGN KEY (department_id) REFERENCES departments(department_id) ); CREATE TABLE tutors ( tutor_id INT PRIMARY KEY AUTO_INCREMENT, employee_id VARCHAR(20) UNIQUE NOT NULL, first_name VARCHAR(50) NOT NULL, last_name VARCHAR(50) NOT NULL, specialization VARCHAR(100), department_id INT, designation VARCHAR(50), FOREIGN KEY (department_id) REFERENCES departments(department_id) );

Advanced Features

-- Stored Procedure for Student Enrollment CREATE PROCEDURE enroll_student( IN p_student_id INT, IN p_offering_id INT, OUT p_status VARCHAR(50) ) BEGIN -- Validation logic for enrollment DECLARE v_current_enrollment INT; DECLARE v_max_enrollment INT; SELECT co.current_enrollment, c.max_enrollment INTO v_current_enrollment, v_max_enrollment FROM course_offerings co JOIN courses c ON co.course_id = c.course_id WHERE co.offering_id = p_offering_id; IF v_current_enrollment >= v_max_enrollment THEN SET p_status = 'Course is full'; ELSE -- Enroll student logic INSERT INTO enrollments (student_id, offering_id, enrollment_date) VALUES (p_student_id, p_offering_id, CURDATE()); SET p_status = 'Successfully enrolled'; END IF; END; -- Trigger for Automatic CGPA Calculation CREATE TRIGGER update_student_cgpa AFTER UPDATE ON enrollments FOR EACH ROW BEGIN -- Recalculate CGPA based on completed courses UPDATE students SET cgpa = ( SELECT AVG(grade_points) FROM enrollments WHERE student_id = NEW.student_id AND enrollment_status = 'Completed' ) WHERE student_id = NEW.student_id; END;

Analytical Queries

-- Student Performance Analysis SELECT s.registration_number, CONCAT(s.first_name, ' ', s.last_name) AS student_name, c.course_code, c.course_name, ROUND(SUM((ar.score / a.max_score) * a.weightage), 2) AS final_grade FROM students s JOIN enrollments e ON s.student_id = e.student_id JOIN course_offerings co ON e.offering_id = co.offering_id JOIN courses c ON co.course_id = c.course_id JOIN assessments a ON co.offering_id = a.offering_id JOIN assessment_results ar ON a.assessment_id = ar.assessment_id GROUP BY s.student_id, c.course_id;

Implementation Highlights

  • Normalized database design (3NF)
  • Complex relationships handled through junction tables
  • Automated grade calculation with triggers
  • Capacity management with stored procedures
  • Performance optimized with strategic indexing
  • Comprehensive views for reporting
