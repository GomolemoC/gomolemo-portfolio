University Database System - MySQL

Complete database schema and queries

database.sql 
-- Create Database
CREATE DATABASE university_management;
USE university_management;

-- Departments Table
CREATE TABLE departments (
    department_id INT PRIMARY KEY AUTO_INCREMENT,
    department_code VARCHAR(10) UNIQUE NOT NULL,
    department_name VARCHAR(100) NOT NULL,
    head_of_department INT,
    established_year YEAR,
    contact_email VARCHAR(100),
    contact_phone VARCHAR(20)
);

-- Students Table
CREATE TABLE students (
    student_id INT PRIMARY KEY AUTO_INCREMENT,
    registration_number VARCHAR(20) UNIQUE NOT NULL,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    date_of_birth DATE NOT NULL,
    gender ENUM('Male', 'Female', 'Other') NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    phone VARCHAR(20),
    address TEXT,
    enrollment_date DATE NOT NULL,
    expected_graduation_date DATE,
    current_semester INT DEFAULT 1,
    department_id INT,
    status ENUM('Active', 'Inactive', 'Graduated', 'Suspended') DEFAULT 'Active',
    cgpa DECIMAL(4,2) DEFAULT 0.00,
    FOREIGN KEY (department_id) REFERENCES departments(department_id)
);

-- Courses Table
CREATE TABLE courses (
    course_id INT PRIMARY KEY AUTO_INCREMENT,
    course_code VARCHAR(15) UNIQUE NOT NULL,
    course_name VARCHAR(100) NOT NULL,
    description TEXT,
    credits INT NOT NULL CHECK (credits > 0 AND credits <= 6),
    course_level ENUM('Undergraduate', 'Graduate', 'Postgraduate') DEFAULT 'Undergraduate',
    department_id INT,
    prerequisite_course_id INT,
    max_enrollment INT DEFAULT 60,
    FOREIGN KEY (department_id) REFERENCES departments(department_id),
    FOREIGN KEY (prerequisite_course_id) REFERENCES courses(course_id)
);

-- Tutors Table
CREATE TABLE tutors (
    tutor_id INT PRIMARY KEY AUTO_INCREMENT,
    employee_id VARCHAR(20) UNIQUE NOT NULL,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    specialization VARCHAR(100),
    department_id INT,
    designation VARCHAR(50),
    FOREIGN KEY (department_id) REFERENCES departments(department_id)
);

-- Course Offerings Table
CREATE TABLE course_offerings (
    offering_id INT PRIMARY KEY AUTO_INCREMENT,
    course_id INT NOT NULL,
    semester VARCHAR(20) NOT NULL,
    tutor_id INT,
    section_number VARCHAR(5) DEFAULT '01',
    schedule_day VARCHAR(20),
    schedule_time TIME,
    room_number VARCHAR(20),
    current_enrollment INT DEFAULT 0,
    FOREIGN KEY (course_id) REFERENCES courses(course_id),
    FOREIGN KEY (tutor_id) REFERENCES tutors(tutor_id)
);

-- Enrollments Table
CREATE TABLE enrollments (
    enrollment_id INT PRIMARY KEY AUTO_INCREMENT,
    student_id INT NOT NULL,
    offering_id INT NOT NULL,
    enrollment_date DATE NOT NULL,
    status ENUM('Enrolled', 'Waitlisted', 'Dropped', 'Completed') DEFAULT 'Enrolled',
    grade VARCHAR(3),
    FOREIGN KEY (student_id) REFERENCES students(student_id),
    FOREIGN KEY (offering_id) REFERENCES course_offerings(offering_id)
);

-- Sample Queries
-- Get student performance
SELECT 
    s.student_id,
    s.registration_number,
    CONCAT(s.first_name, ' ', s.last_name) AS full_name,
    d.department_name,
    s.cgpa,
    COUNT(e.enrollment_id) AS courses_taken
FROM students s
JOIN departments d ON s.department_id = d.department_id
LEFT JOIN enrollments e ON s.student_id = e.student_id
GROUP BY s.student_id
HAVING s.cgpa >= 3.5
ORDER BY s.cgpa DESC;

-- Stored Procedure for enrollment
DELIMITER //
CREATE PROCEDURE enroll_student(
    IN p_student_id INT,
    IN p_offering_id INT,
    OUT p_status VARCHAR(50)
)
BEGIN
    DECLARE v_current_enrollment INT;
    DECLARE v_max_enrollment INT;
    
    SELECT co.current_enrollment, c.max_enrollment 
    INTO v_current_enrollment, v_max_enrollment
    FROM course_offerings co
    JOIN courses c ON co.course_id = c.course_id
    WHERE co.offering_id = p_offering_id;
    
    IF v_current_enrollment >= v_max_enrollment THEN
        SET p_status = 'Course is full';
    ELSE
        INSERT INTO enrollments (student_id, offering_id, enrollment_date)
        VALUES (p_student_id, p_offering_id, CURDATE());
        SET p_status = 'Successfully enrolled';
    END IF;
END //
DELIMITER ;

Healthcare Data Analyzer - Python

Complete Python application for healthcare data analysis

healthcare_analyzer.py 
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from datetime import datetime

class HealthcareAnalyzer:
    def __init__(self):
        self.patient_data = None
        self.analysis_results = {}
    
    def load_data(self, filepath):
        """Load patient data from CSV file"""
        try:
            self.patient_data = pd.read_csv(filepath)
            print(f"Loaded {len(self.patient_data)} patient records")
            return True
        except Exception as e:
            print(f"Error loading data: {e}")
            return False
    
    def analyze_patient_risk(self):
        """Analyze patient risk factors"""
        if self.patient_data is None:
            print("No data loaded")
            return
        
        risk_scores = []
        
        for _, patient in self.patient_data.iterrows():
            score = 0
            
            # Age factor
            if patient.get('age', 0) > 65:
                score += 3
            elif patient.get('age', 0) > 45:
                score += 2
            
            # Blood pressure
            if patient.get('blood_pressure', 0) > 140:
                score += 3
            elif patient.get('blood_pressure', 0) > 120:
                score += 2
            
            # Cholesterol
            if patient.get('cholesterol', 0) > 240:
                score += 3
            elif patient.get('cholesterol', 0) > 200:
                score += 2
            
            # BMI
            bmi = patient.get('bmi', 0)
            if bmi > 30:
                score += 3
            elif bmi > 25:
                score += 2
            
            risk_scores.append(score)
        
        self.patient_data['risk_score'] = risk_scores
        self.patient_data['risk_level'] = pd.cut(
            risk_scores,
            bins=[-1, 3, 6, 10, 20],
            labels=['Low', 'Medium', 'High', 'Critical']
        )
        
        return self.patient_data['risk_level'].value_counts()
    
    def generate_health_trends(self):
        """Generate health trend analysis"""
        if self.patient_data is None:
            print("No data loaded")
            return
        
        trends = {
            'average_age': self.patient_data['age'].mean(),
            'average_bmi': self.patient_data['bmi'].mean(),
            'high_blood_pressure_pct': (self.patient_data['blood_pressure'] > 140).mean() * 100,
            'high_cholesterol_pct': (self.patient_data['cholesterol'] > 240).mean() * 100,
            'obesity_pct': (self.patient_data['bmi'] > 30).mean() * 100
        }
        
        return trends
    
    def create_visualizations(self):
        """Create data visualizations"""
        if self.patient_data is None:
            print("No data loaded")
            return
        
        fig, axes = plt.subplots(2, 2, figsize=(12, 10))
        
        # Age distribution
        axes[0, 0].hist(self.patient_data['age'], bins=20, color='skyblue')
        axes[0, 0].set_title('Age Distribution')
        
        # BMI distribution
        axes[0, 1].hist(self.patient_data['bmi'], bins=20, color='lightgreen')
        axes[0, 1].set_title('BMI Distribution')
        
        # Risk levels
        risk_counts = self.patient_data['risk_level'].value_counts()
        axes[1, 0].bar(risk_counts.index, risk_counts.values, color='salmon')
        axes[1, 0].set_title('Risk Level Distribution')
        
        # Blood pressure vs Age
        axes[1, 1].scatter(
            self.patient_data['age'],
            self.patient_data['blood_pressure'],
            alpha=0.5,
            color='purple'
        )
        axes[1, 1].set_title('Blood Pressure vs Age')
        
        plt.tight_layout()
        plt.savefig('healthcare_analysis.png')
        plt.show()
    
    def export_results(self, filename='healthcare_results.csv'):
        """Export analysis results to CSV"""
        if self.patient_data is not None:
            self.patient_data.to_csv(filename, index=False)
            print(f"Results exported to {filename}")

def main():
    analyzer = HealthcareAnalyzer()
    
    # Load data
    if analyzer.load_data('patient_data.csv'):
        # Analyze risk
        print("\n=== PATIENT RISK ANALYSIS ===")
        risk_distribution = analyzer.analyze_patient_risk()
        print(risk_distribution)
        
        # Generate trends
        print("\n=== HEALTH TRENDS ===")
        trends = analyzer.generate_health_trends()
        for key, value in trends.items():
            print(f"{key.replace('_', ' ').title()}: {value:.2f}")
        
        # Create visualizations
        print("\nCreating visualizations...")
        analyzer.create_visualizations()
        
        # Export results
        analyzer.export_results()

if __name__ == "__main__":
    main()

Password Security Checker - Java

Java application for analyzing password strength

PasswordChecker.java 
import java.util.Scanner;
import java.util.regex.Pattern;
import java.util.regex.Matcher;

public class PasswordChecker {
    private String password;
    private int score;
    private StringBuilder feedback;
    
    public PasswordChecker(String password) {
        this.password = password;
        this.score = 0;
        this.feedback = new StringBuilder();
    }
    
    public void analyzePassword() {
        // Length check
        if (password.length() >= 12) {
            score += 3;
            feedback.append("✓ Excellent length (12+ characters)\n");
        } else if (password.length() >= 8) {
            score += 2;
            feedback.append("✓ Good length (8-11 characters)\n");
        } else if (password.length() >= 6) {
            score += 1;
            feedback.append("⚠ Fair length (6-7 characters)\n");
        } else {
            feedback.append("✗ Too short (less than 6 characters)\n");
        }
        
        // Uppercase letters
        if (Pattern.compile("[A-Z]").matcher(password).find()) {
            score += 2;
            feedback.append("✓ Contains uppercase letters\n");
        } else {
            feedback.append("✗ No uppercase letters\n");
        }
        
        // Lowercase letters
        if (Pattern.compile("[a-z]").matcher(password).find()) {
            score += 2;
            feedback.append("✓ Contains lowercase letters\n");
        } else {
            feedback.append("✗ No lowercase letters\n");
        }
        
        // Numbers
        if (Pattern.compile("[0-9]").matcher(password).find()) {
            score += 2;
            feedback.append("✓ Contains numbers\n");
        } else {
            feedback.append("✗ No numbers\n");
        }
        
        // Special characters
        if (Pattern.compile("[^a-zA-Z0-9]").matcher(password).find()) {
            score += 3;
            feedback.append("✓ Contains special characters\n");
        } else {
            feedback.append("⚠ No special characters\n");
        }
        
        // Common patterns check
        if (Pattern.compile("(.)\\1{2,}").matcher(password).find()) {
            score -= 2;
            feedback.append("✗ Contains repeated characters\n");
        }
        
        if (Pattern.compile("(123|abc|password|qwerty)").matcher(password.toLowerCase()).find()) {
            score -= 3;
            feedback.append("✗ Contains common patterns\n");
        }
        
        // Clamp score between 0 and 12
        score = Math.max(0, Math.min(12, score));
    }
    
    public String getStrengthLevel() {
        if (score >= 10) {
            return "STRONG";
        } else if (score >= 7) {
            return "GOOD";
        } else if (score >= 4) {
            return "FAIR";
        } else {
            return "WEAK";
        }
    }
    
    public int getScore() {
        return score;
    }
    
    public String getFeedback() {
        return feedback.toString();
    }
    
    public String getRecommendations() {
        StringBuilder recommendations = new StringBuilder();
        
        if (password.length() < 12) {
            recommendations.append("• Use at least 12 characters\n");
        }
        if (!Pattern.compile("[A-Z]").matcher(password).find()) {
            recommendations.append("• Add uppercase letters\n");
        }
        if (!Pattern.compile("[0-9]").matcher(password).find()) {
            recommendations.append("• Add numbers\n");
        }
        if (!Pattern.compile("[^a-zA-Z0-9]").matcher(password).find()) {
            recommendations.append("• Add special characters (!@#$%^&*)\n");
        }
        
        return recommendations.toString();
    }
    
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        
        System.out.println("=== PASSWORD SECURITY CHECKER ===");
        System.out.print("Enter password to check: ");
        String password = scanner.nextLine();
        
        PasswordChecker checker = new PasswordChecker(password);
        checker.analyzePassword();
        
        System.out.println("\n=== ANALYSIS RESULTS ===");
        System.out.println("Password Length: " + password.length());
        System.out.println("Security Score: " + checker.getScore() + "/12");
        System.out.println("Strength Level: " + checker.getStrengthLevel());
        
        System.out.println("\n=== DETAILED FEEDBACK ===");
        System.out.println(checker.getFeedback());
        
        if (checker.getScore() < 10) {
            System.out.println("=== RECOMMENDATIONS ===");
            System.out.println(checker.getRecommendations());
        }
        
        scanner.close();
    }
}

Network Security Scanner - Python

Python tool for network scanning and security analysis

security_scanner.py 
import socket
import subprocess
import sys
from datetime import datetime
import ipaddress

class NetworkSecurityScanner:
    def __init__(self):
        self.results = []
    
    def scan_port(self, host, port):
        """Scan a single port on a host"""
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.settimeout(1)
        
        try:
            result = sock.connect_ex((host, port))
            if result == 0:
                return True
            else:
                return False
        except socket.error:
            return False
        finally:
            sock.close()
    
    def scan_ports(self, host, start_port=1, end_port=1024):
        """Scan a range of ports on a host"""
        print(f"\nScanning {host} (ports {start_port}-{end_port})")
        print("=" * 50)
        
        open_ports = []
        
        for port in range(start_port, end_port + 1):
            if self.scan_port(host, port):
                service = self.get_service_name(port)
                print(f"Port {port}: OPEN - {service}")
                open_ports.append({
                    'port': port,
                    'service': service,
                    'status': 'open'
                })
        
        return open_ports
    
    def get_service_name(self, port):
        """Get service name for common ports"""
        common_ports = {
            20: 'FTP-Data',
            21: 'FTP',
            22: 'SSH',
            23: 'Telnet',
            25: 'SMTP',
            53: 'DNS',
            80: 'HTTP',
            110: 'POP3',
            143: 'IMAP',
            443: 'HTTPS',
            3306: 'MySQL',
            3389: 'RDP',
            5432: 'PostgreSQL',
            8080: 'HTTP-Proxy'
        }
        return common_ports.get(port, 'Unknown')
    
    def scan_network(self, network):
        """Scan entire network for active hosts"""
        print(f"\nScanning network: {network}")
        print("=" * 50)
        
        active_hosts = []
        
        try:
            # Ping sweep
            for ip in ipaddress.IPv4Network(network):
                ip_str = str(ip)
                if self.ping_host(ip_str):
                    print(f"Host active: {ip_str}")
                    active_hosts.append(ip_str)
        except ValueError as e:
            print(f"Invalid network: {e}")
        
        return active_hosts
    
    def ping_host(self, host):
        """Ping a host to check if it's active"""
        try:
            if sys.platform == "win32":
                result = subprocess.run(
                    ['ping', '-n', '1', '-w', '1000', host],
                    capture_output=True,
                    text=True
                )
            else:
                result = subprocess.run(
                    ['ping', '-c', '1', '-W', '1', host],
                    capture_output=True,
                    text=True
                )
            
            return result.returncode == 0
        except Exception:
            return False
    
    def analyze_security(self, open_ports):
        """Analyze security implications of open ports"""
        print("\n=== SECURITY ANALYSIS ===")
        print("=" * 50)
        
        security_issues = []
        
        for port_info in open_ports:
            port = port_info['port']
            service = port_info['service']
            
            if port in [21, 23, 3389]:
                security_issues.append(f"⚠ {service} (Port {port}) - Insecure protocol, consider disabling")
            elif port in [80, 8080]:
                security_issues.append(f"⚠ {service} (Port {port}) - Ensure HTTPS is configured")
            elif port == 3306:
                security_issues.append(f"⚠ MySQL (Port {port}) - Database should not be publicly accessible")
            elif port == 22:
                security_issues.append(f"ℹ SSH (Port {port}) - Ensure key-based authentication")
        
        if security_issues:
            for issue in security_issues:
                print(issue)
        else:
            print("✓ No critical security issues found")
        
        return security_issues
    
    def generate_report(self, host, open_ports):
        """Generate security report"""
        report = {
            'timestamp': datetime.now().isoformat(),
            'target': host,
            'open_ports': open_ports,
            'total_open_ports': len(open_ports)
        }
        
        return report

def main():
    scanner = NetworkSecurityScanner()
    
    print("=== NETWORK SECURITY SCANNER ===")
    print("This tool is for educational purposes only")
    print("Always get permission before scanning networks\n")
    
    while True:
        print("\nOptions:")
        print("1. Scan a single host")
        print("2. Scan a network range")
        print("3. Exit")
        
        choice = input("\nEnter choice: ")
        
        if choice == '1':
            host = input("Enter host to scan (e.g., localhost): ")
            try:
                start_port = int(input("Enter start port (default 1): ") or "1")
                end_port = int(input("Enter end port (default 1024): ") or "1024")
                
                open_ports = scanner.scan_ports(host, start_port, end_port)
                
                if open_ports:
                    scanner.analyze_security(open_ports)
                    report = scanner.generate_report(host, open_ports)
                    print(f"\nTotal open ports: {report['total_open_ports']}")
                else:
                    print("\nNo open ports found")
                    
            except ValueError:
                print("Invalid port number")
        
        elif choice == '2':
            network = input("Enter network (e.g., 192.168.1.0/24): ")
            active_hosts = scanner.scan_network(network)
            print(f"\nFound {len(active_hosts)} active hosts")
        
        elif choice == '3':
            print("\nGoodbye!")
            break
        
        else:
            print("Invalid choice")

if __name__ == "__main__":
    main()
